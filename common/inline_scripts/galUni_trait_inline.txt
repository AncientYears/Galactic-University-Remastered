potential = {
	from = { has_country_flag = galUni_hiring_phase_two }
}
allow = { always = yes }
effect = {
	if = { limit = { has_global_flag = galUni_storage_created }
		event_target:galUni_storage = {
			every_owned_leader = {
				limit = { has_leader_flag = galUni_preview }
				tooltip = { add_trait = $trait_tool$ }
			}
		}
	}
	custom_tooltip = "galUni_trait_tooltip_separator"
	custom_tooltip = "$trait_tool$_desc"
	hidden_effect = {
		if = { limit = { from = { has_country_flag = $flag$ } }
			root = { galUni_trait_enable_disable_effect = { EFFECT = remove_country_flag TYPE = $type$ NAME = $trait_name$ } }
			root = { subtract_variable = { which = galUni_hiringCost value = $cost$ } }
			root = { set_variable = { which = $trait_id$ value = 0 } }
		} else = {
			root = { galUni_trait_enable_disable_effect = { EFFECT = set_country_flag TYPE = $type$ NAME = $trait_name$ } }
			root = { change_variable = { which = galUni_hiringCost value = $cost$ } }
			root = { set_variable = { which = $trait_id$ value = 1 } }
		}
	}
}
